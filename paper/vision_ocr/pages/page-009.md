\[
\begin{bmatrix}
Q^{cq}_{\text{low}} \\[1ex]
Q^{cq}_{\text{high}}
\end{bmatrix}
\simeq
\begin{bmatrix}
\cos{\theta^{cq}_{NL}} & \sin{\theta^{cq}_{NL}} \\
-\sin{\theta^{cq}_{NL}} & \cos{\theta^{cq}_{NL}}
\end{bmatrix}
\begin{bmatrix}

\end{bmatrix}
\]
